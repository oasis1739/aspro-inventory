// 기존 Code.gs 맨 아래에 추가 (전체 교체)
function doGet(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  // 매핑
  let sh = ss.getSheetByName('aspro_match');
  if (!sh) { sh = ss.insertSheet('aspro_match'); sh.appendRow(['사방넷코드','김포물류품번','메모','등록일']); }
  const map = {};
  sh.getDataRange().getValues().slice(1).forEach(r => { if (r[0]) map[String(r[0])] = String(r[1]); });
  // 품절
  let so = ss.getSheetByName('aspro_soldout');
  if (!so) { so = ss.insertSheet('aspro_soldout'); so.appendRow(['사방넷코드','등록일']); }
  const soldout = {};
  so.getDataRange().getValues().slice(1).forEach(r => { if (r[0]) soldout[String(r[0])] = true; });
  // 신원재고
  let kc = ss.getSheetByName('aspro_kcfi');
  if (!kc) { kc = ss.insertSheet('aspro_kcfi'); kc.appendRow(['사방넷코드','상품명','타입','색상','사이즈','재고','업데이트일','비고']); }
  const kcfi = {};
  kc.getDataRange().getValues().slice(1).forEach(r => {
    const sc = String(r[0] || '').trim();
    const qty = parseInt(r[5], 10);
    if (sc && !isNaN(qty)) kcfi[sc] = qty;
  });
  // 강제품절
  let fz = ss.getSheetByName('aspro_forcezero');
  if (!fz) { fz = ss.insertSheet('aspro_forcezero'); fz.appendRow(['사방넷코드','사유','등록일']); }
  const forcezero = {};
  fz.getDataRange().getValues().slice(1).forEach(r => {
    if (r[0]) forcezero[String(r[0])] = { reason: String(r[1]||''), date: String(r[2]||'') };
  });
  return ContentService.createTextOutput(JSON.stringify({ok:true, map, soldout, kcfi, forcezero}))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const body = JSON.parse(e.postData.contents);
  // 매핑 저장
  if (body.map) {
    let sh = ss.getSheetByName('aspro_match');
    if (!sh) { sh = ss.insertSheet('aspro_match'); }
    sh.clearContents();
    sh.appendRow(['사방넷코드','김포물류품번','메모','등록일']);
    const rows = Object.entries(body.map).map(([k,v]) => [k, v, '', new Date().toLocaleDateString('ko-KR')]);
    if (rows.length) sh.getRange(2,1,rows.length,4).setValues(rows);
  }
  // 품절 저장
  if (body.soldout) {
    let so = ss.getSheetByName('aspro_soldout');
    if (!so) { so = ss.insertSheet('aspro_soldout'); }
    so.clearContents();
    so.appendRow(['사방넷코드','등록일']);
    const rows = Object.entries(body.soldout).map(([k,v]) => [k, v]);
    if (rows.length) so.getRange(2,1,rows.length,2).setValues(rows);
  }
  // 강제품절 저장
  if (body.forcezero !== undefined) {
    let fz = ss.getSheetByName('aspro_forcezero');
    if (!fz) { fz = ss.insertSheet('aspro_forcezero'); }
    fz.clearContents();
    fz.appendRow(['사방넷코드','사유','등록일']);
    const rows = Object.entries(body.forcezero).map(([k,v]) => [k, v.reason||'', v.date||'']);
    if (rows.length) fz.getRange(2,1,rows.length,3).setValues(rows);
  }
  // 신원재고 저장
  if (body.kcfi) {
    let kc = ss.getSheetByName('aspro_kcfi');
    if (!kc) { kc = ss.insertSheet('aspro_kcfi'); }
    kc.clearContents();
    kc.appendRow(['사방넷코드','재고','업데이트일']);
    const rows = Object.entries(body.kcfi).map(([k,v]) => [k, v, new Date().toLocaleDateString('ko-KR')]);
    if (rows.length) kc.getRange(2,1,rows.length,3).setValues(rows);
  }
  const saved = (body.map ? Object.keys(body.map).length : 0);
  return ContentService.createTextOutput(JSON.stringify({ok:true, saved}))
    .setMimeType(ContentService.MimeType.JSON);
}
