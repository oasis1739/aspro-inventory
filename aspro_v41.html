function doGet(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sh = ss.getSheetByName('aspro_match');
  if (!sh) { sh = ss.insertSheet('aspro_match'); sh.appendRow(['사방넷코드','김포물류품번','메모','등록일']); }
  const data = sh.getDataRange().getValues();
  const map = {};
  data.slice(1).forEach(r => { if (r[0]) map[String(r[0])] = String(r[1]); });
  let so = ss.getSheetByName('aspro_soldout');
  if (!so) { so = ss.insertSheet('aspro_soldout'); so.appendRow(['사방넷코드','등록일']); }
  const soData = so.getDataRange().getValues();
  const soldout = {};
  soData.slice(1).forEach(r => { if (r[0]) soldout[String(r[0])] = true; });
  let kc = ss.getSheetByName('aspro_kcfi');
  if (!kc) { kc = ss.insertSheet('aspro_kcfi'); kc.appendRow(['사방넷코드','상품명','타입','색상','사이즈','재고','업데이트일','비고']); }
  const kcData = kc.getDataRange().getValues();
  const kcfi = {};
  kcData.slice(1).forEach(r => {
    const sc = String(r[0] || '').trim();
    const qty = parseInt(r[5], 10);
    if (sc && !isNaN(qty)) kcfi[sc] = qty;
  });
  return ContentService.createTextOutput(JSON.stringify({ok:true, map, soldout, kcfi}))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const body = JSON.parse(e.postData.contents);
  let sh = ss.getSheetByName('aspro_match');
  if (!sh) { sh = ss.insertSheet('aspro_match'); sh.appendRow(['사방넷코드','김포물류품번','메모','등록일']); }
  const existing = sh.getDataRange().getValues();
  const rowIdx = {};
  existing.slice(1).forEach((r,i) => { if (r[0]) rowIdx[String(r[0])] = i+2; });
  let saved = 0;
  const today = new Date().toLocaleDateString('ko-KR');
  Object.entries(body.map || {}).forEach(([sc, gp]) => {
    if (rowIdx[sc]) sh.getRange(rowIdx[sc],2).setValue(gp);
    else { sh.appendRow([sc, gp, '', today]); saved++; }
  });
  if (body.soldout) {
    let so = ss.getSheetByName('aspro_soldout');
    if (!so) { so = ss.insertSheet('aspro_soldout'); so.appendRow(['사방넷코드','등록일']); }
    const soData = so.getDataRange().getValues();
    const soIdx = {};
    soData.slice(1).forEach((r,i) => { if (r[0]) soIdx[String(r[0])] = i+2; });
    Object.keys(body.soldout).forEach(sc => { if (!soIdx[sc]) so.appendRow([sc, today]); });
    if (body.removed_soldout) {
      const soAll = so.getDataRange().getValues();
      const toRemove = new Set(body.removed_soldout);
      for (let i = soAll.length; i >= 2; i--) {
        if (toRemove.has(String(soAll[i-1][0]))) so.deleteRow(i);
      }
    }
  }
  if (body.kcfi) {
    let kc = ss.getSheetByName('aspro_kcfi');
    if (!kc) { kc = ss.insertSheet('aspro_kcfi'); kc.appendRow(['사방넷코드','상품명','타입','색상','사이즈','재고','업데이트일','비고']); }
    const kcData = kc.getDataRange().getValues();
    const kcIdx = {};
    kcData.slice(1).forEach((r,i) => { if (r[0]) kcIdx[String(r[0])] = i+2; });
    Object.entries(body.kcfi).forEach(([sc, qty]) => {
      if (kcIdx[sc]) {
        kc.getRange(kcIdx[sc], 6).setValue(qty);
        kc.getRange(kcIdx[sc], 7).setValue(today);
      }
    });
  }
  return ContentService.createTextOutput(JSON.stringify({ok:true, saved}))
    .setMimeType(ContentService.MimeType.JSON);
}
