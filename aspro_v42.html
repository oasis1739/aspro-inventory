function doGet(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const type = (e && e.parameter && e.parameter.type) || 'base';

  // type=raw: 원본 데이터만 반환 (사방넷/이카운트/김포) — 별도 요청으로 분리
  if (type === 'raw') {
    let sabRawData = [];
    const sabSh = ss.getSheetByName('사방넷단품원본');
    if (sabSh) sabRawData = sabSh.getDataRange().getValues();

    let icRaw = [];
    const icSh = ss.getSheetByName('이카운트원본');
    if (icSh) icRaw = icSh.getDataRange().getValues();

    let gpRaw = [];
    const gpSh = ss.getSheetByName('김포원본');
    if (gpSh) gpRaw = gpSh.getDataRange().getValues();

    return ContentService.createTextOutput(JSON.stringify({ok:true, sabRawData, icRaw, gpRaw}))
      .setMimeType(ContentService.MimeType.JSON);
  }

  // type=base (기본): 매핑/품절/kcfi 반환
  let sh = ss.getSheetByName('aspro_match');
  if (!sh) { sh = ss.insertSheet('aspro_match'); sh.appendRow(['사방넷코드','김포물류품번','상품명','메모','등록일']); }
  const data = sh.getDataRange().getValues();
  const map = {};
  data.slice(1).forEach(r => { if (r[0]) map[String(r[0]).trim()] = String(r[1]).trim(); });

  let so = ss.getSheetByName('aspro_soldout');
  if (!so) { so = ss.insertSheet('aspro_soldout'); so.appendRow(['사방넷코드','등록일']); }
  const soData = so.getDataRange().getValues();
  const soldout = {};
  soData.slice(1).forEach(r => {
    const sc = String(r[0] || '').trim();
    if (sc) soldout[sc] = String(r[1] || new Date().toLocaleDateString('ko-KR'));
  });

  let kc = ss.getSheetByName('aspro_kcfi');
  if (!kc) { kc = ss.insertSheet('aspro_kcfi'); kc.appendRow(['사방넷코드','재고','업데이트일']); }
  const kcData = kc.getDataRange().getValues();
  const kcfi = {};
  if (kcData.length > 1) {
    // 헤더에서 '재고' 컬럼 인덱스를 동적으로 찾음 (컬럼 구조 변경에 강건)
    const kcHeader = kcData[0].map(h => String(h).trim());
    const qtyIdx = kcHeader.indexOf('재고');
    const useIdx = qtyIdx >= 0 ? qtyIdx : 1; // 못 찾으면 col[1] 기본값
    kcData.slice(1).forEach(r => {
      const sc = String(r[0] || '').trim();
      const qty = parseInt(r[useIdx], 10);
      if (sc && !isNaN(qty)) kcfi[sc] = qty;
    });
  }

  let nn = ss.getSheetByName('aspro_normname');
  if (!nn) { nn = ss.insertSheet('aspro_normname'); nn.appendRow(['정규화상품명(normName)','물류품번','원본사방넷상품명','등록일']); }
  const nnData = nn.getDataRange().getValues();
  const normname = {};
  nnData.slice(1).forEach(r => { if (r[0] && r[1]) normname[String(r[0])] = String(r[1]); });

  let ic = ss.getSheetByName('aspro_iccodemap');
  if (!ic) { ic = ss.insertSheet('aspro_iccodemap'); ic.appendRow(['사방넷코드','이카운트품목코드','등록일']); }
  const icData = ic.getDataRange().getValues();
  const iccodemap = {};
  icData.slice(1).forEach(r => { if (r[0] && r[1]) iccodemap[String(r[0]).trim()] = String(r[1]).trim(); });

  // aspro_setmap: 세트규칙 (A=자체코드prefix, B=김포prefix, C=색상매핑)
  // 예: JW25SS02 | 252JW19 | 아이보리:IV,스카이블루:BL,블랙:BK
  const smSh = ss.getSheetByName('aspro_setmap');
  const setmap = {};
  if (smSh && smSh.getLastRow() > 1) {
    smSh.getDataRange().getValues().slice(1).forEach(r => {
      const prefix = String(r[0]||'').trim(), gpPfx = String(r[1]||'').trim(), colorStr = String(r[2]||'').trim();
      if (!prefix || !gpPfx) return;
      const colorMap = {};
      colorStr.split(',').forEach(p => { const [k,v] = p.split(':'); if(k&&v) colorMap[k.trim()]=v.trim(); });
      setmap[prefix] = { gpPrefix: gpPfx, colorMap };
    });
  }

  return ContentService.createTextOutput(JSON.stringify({ok:true, map, soldout, kcfi, normname, iccodemap, setmap}))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const payload = JSON.parse(e.postData.contents);
    const now = new Date().toLocaleDateString('ko-KR');

    if (payload.map) {
      let sh = ss.getSheetByName('aspro_match');
      if (!sh) { sh = ss.insertSheet('aspro_match'); }
      sh.clearContents();
      const nmMap = payload.nameMap || {};
      const rows = [['사방넷코드','김포물류품번','상품명','메모','등록일']];
      Object.entries(payload.map).forEach(([k,v]) => { rows.push([k, v, nmMap[k]||'', '', now]); });
      sh.getRange(1, 1, rows.length, 5).setValues(rows);
    }

    if (Object.prototype.hasOwnProperty.call(payload, 'soldout')) {
      let so = ss.getSheetByName('aspro_soldout');
      if (!so) { so = ss.insertSheet('aspro_soldout'); }
      so.clearContents();
      const rows = [['사방넷코드','등록일']];
      Object.entries(payload.soldout).forEach(([k,v]) => { rows.push([k, v === true ? now : String(v)]); });
      so.getRange(1, 1, rows.length, 2).setValues(rows);
    }

    if (payload.kcfi && Object.keys(payload.kcfi).length > 0) {
      let kc = ss.getSheetByName('aspro_kcfi');
      if (!kc) { kc = ss.insertSheet('aspro_kcfi'); }
      kc.clearContents();
      const rows = [['사방넷코드','재고','업데이트일']];
      Object.entries(payload.kcfi).forEach(([k,v]) => { rows.push([k, v, now]); });
      kc.getRange(1, 1, rows.length, 3).setValues(rows);
    }

    if (payload.normname && Object.keys(payload.normname).length > 0) {
      let nn = ss.getSheetByName('aspro_normname');
      if (!nn) { nn = ss.insertSheet('aspro_normname'); }
      nn.clearContents();
      const rows = [['정규화상품명(normName)','물류품번','원본사방넷상품명','등록일']];
      Object.entries(payload.normname).forEach(([k,v]) => { rows.push([k, v, '', now]); });
      nn.getRange(1, 1, rows.length, 4).setValues(rows);
    }

    if (payload.iccodemap && Object.keys(payload.iccodemap).length > 0) {
      let ic = ss.getSheetByName('aspro_iccodemap');
      if (!ic) { ic = ss.insertSheet('aspro_iccodemap'); }
      ic.clearContents();
      const rows = [['사방넷코드','이카운트품목코드','등록일']];
      Object.entries(payload.iccodemap).forEach(([k,v]) => { rows.push([k, v, now]); });
      ic.getRange(1, 1, rows.length, 3).setValues(rows);
    }

    const saved = payload.map ? Object.keys(payload.map).length : 0;
    return ContentService.createTextOutput(JSON.stringify({ok:true, saved}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ok:false, error: err.message}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
