function doGet(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const type = (e && e.parameter && e.parameter.type) || 'base';

  // type=raw: 원본 데이터만 반환 (사방넷/이카운트/김포) — 별도 요청으로 분리
  if (type === 'raw') {
    let sabRawData = [];
    const sabSh = ss.getSheetByName('사방넷단품원본');
    if (sabSh) sabRawData = sabSh.getDataRange().getValues();

    let icRaw = [];
    const icSh = ss.getSheetByName('이카운트원본');
    if (icSh) icRaw = icSh.getDataRange().getValues();

    let gpRaw = [];
    const gpSh = ss.getSheetByName('김포원본');
    if (gpSh) gpRaw = gpSh.getDataRange().getValues();

    return ContentService.createTextOutput(JSON.stringify({ok:true, sabRawData, icRaw, gpRaw}))
      .setMimeType(ContentService.MimeType.JSON);
  }

  // type=base (기본): 매핑/품절/kcfi 반환
  let sh = ss.getSheetByName('aspro_match');
  if (!sh) { sh = ss.insertSheet('aspro_match'); sh.appendRow(['사방넷코드','김포물류품번','상품명','메모','등록일']); }
  const data = sh.getDataRange().getValues();
  const map = {};
  data.slice(1).forEach(r => { if (r[0]) map[String(r[0]).trim()] = String(r[1]).trim(); });

  let so = ss.getSheetByName('aspro_soldout');
  if (!so) { so = ss.insertSheet('aspro_soldout'); so.appendRow(['사방넷코드','등록일']); }
  const soData = so.getDataRange().getValues();
  const soldout = {};
  soData.slice(1).forEach(r => {
    const sc = String(r[0] || '').trim();
    if (sc) soldout[sc] = String(r[1] || new Date().toLocaleDateString('ko-KR'));
  });

  let kc = ss.getSheetByName('aspro_kcfi');
  if (!kc) { kc = ss.insertSheet('aspro_kcfi'); kc.appendRow(['사방넷코드','재고','업데이트일']); }
  const kcData = kc.getDataRange().getValues();
  const kcfi = {};
  if (kcData.length > 1) {
    // 헤더에서 '재고' 컬럼 인덱스를 동적으로 찾음 (컬럼 구조 변경에 강건)
    const kcHeader = kcData[0].map(h => String(h).trim());
    const qtyIdx = kcHeader.indexOf('재고');
    const useIdx = qtyIdx >= 0 ? qtyIdx : 1; // 못 찾으면 col[1] 기본값
    kcData.slice(1).forEach(r => {
      const sc = String(r[0] || '').trim();
      const qty = parseInt(r[useIdx], 10);
      if (sc && !isNaN(qty)) kcfi[sc] = qty;
    });
  }

  let nn = ss.getSheetByName('aspro_normname');
  if (!nn) { nn = ss.insertSheet('aspro_normname'); nn.appendRow(['정규화상품명(normName)','물류품번','원본사방넷상품명','등록일']); }
  const nnData = nn.getDataRange().getValues();
  const normname = {};
  nnData.slice(1).forEach(r => { if (r[0] && r[1]) normname[String(r[0])] = String(r[1]); });

  let ic = ss.getSheetByName('aspro_iccodemap');
  if (!ic) { ic = ss.insertSheet('aspro_iccodemap'); ic.appendRow(['사방넷코드','이카운트품목코드','등록일']); }
  const icData = ic.getDataRange().getValues();
  const iccodemap = {};
  icData.slice(1).forEach(r => { if (r[0] && r[1]) iccodemap[String(r[0]).trim()] = String(r[1]).trim(); });


  return ContentService.createTextOutput(JSON.stringify({ok:true, map, soldout, kcfi, normname, iccodemap}))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var data = JSON.parse(e.postData.contents);
    var savedCount = 0;

    // ── aspro_match: UPSERT (신규 삽입 + 기존값 업데이트) ──
    if (data.map && Object.keys(data.map).length > 0) {
      var matchSh = ss.getSheetByName('aspro_match') || ss.insertSheet('aspro_match');
      var lastRow = matchSh.getLastRow();
      var lastCol = Math.max(matchSh.getLastColumn(), 2);
      var allRows = lastRow > 0 ? matchSh.getRange(1, 1, lastRow, lastCol).getValues() : [];
      var rowIdx = {};
      for (var i = 1; i < allRows.length; i++) {
        var c = String(allRows[i][0]).trim();
        if (c) rowIdx[c] = i;
      }
      var toAdd = [];
      for (var sc in data.map) {
        if (!sc) continue;
        var gp = String(data.map[sc]);
        if (rowIdx[sc] !== undefined) { allRows[rowIdx[sc]][1] = gp; }
        else { toAdd.push([sc, gp]); }
        savedCount++;
      }
      if (allRows.length > 0) {
        matchSh.clearContents();
        matchSh.getRange(1, 1, allRows.length, allRows[0].length).setValues(allRows);
      }
      if (toAdd.length > 0) {
        matchSh.getRange(matchSh.getLastRow() + 1, 1, toAdd.length, 2).setValues(toAdd);
      }
    }

    // ── aspro_soldout: 전체 덮어쓰기 (빈 데이터면 스킵) ──
    if (data.soldout && Object.keys(data.soldout).length > 0) {
      var soldoutSh = ss.getSheetByName('aspro_soldout') || ss.insertSheet('aspro_soldout');
      soldoutSh.clearContents();
      var rows = [];
      for (var sc in data.soldout) { if (sc) rows.push([sc, data.soldout[sc] || '']); }
      if (rows.length > 0) soldoutSh.getRange(1, 1, rows.length, 2).setValues(rows);
    }

    // ── aspro_kcfi ──
    if (data.kcfi && Object.keys(data.kcfi).length > 0) {
      var kcfiSh = ss.getSheetByName('aspro_kcfi') || ss.insertSheet('aspro_kcfi');
      kcfiSh.clearContents();
      var rows = [];
      for (var sc in data.kcfi) { if (sc) rows.push([sc, data.kcfi[sc]]); }
      if (rows.length > 0) kcfiSh.getRange(1, 1, rows.length, 2).setValues(rows);
    }

    // ── aspro_normname ──
    if (data.normname && Object.keys(data.normname).length > 0) {
      var nnSh = ss.getSheetByName('aspro_normname') || ss.insertSheet('aspro_normname');
      nnSh.clearContents();
      var rows = [];
      for (var k in data.normname) { if (k) rows.push([k, data.normname[k]]); }
      if (rows.length > 0) nnSh.getRange(1, 1, rows.length, 2).setValues(rows);
    }

    // ── aspro_forcezero ──
    if (data.forcezero && Object.keys(data.forcezero).length > 0) {
      var fzSh = ss.getSheetByName('aspro_forcezero') || ss.insertSheet('aspro_forcezero');
      fzSh.clearContents();
      var rows = [];
      for (var c in data.forcezero) { if (c) rows.push([c]); }
      if (rows.length > 0) fzSh.getRange(1, 1, rows.length, 1).setValues(rows);
    }

    // ── aspro_iccodemap ──
    if (data.iccodemap && Object.keys(data.iccodemap).length > 0) {
      var icSh = ss.getSheetByName('aspro_iccodemap') || ss.insertSheet('aspro_iccodemap');
      icSh.clearContents();
      var rows = [];
      for (var sc in data.iccodemap) { if (sc) rows.push([sc, data.iccodemap[sc]]); }
      if (rows.length > 0) icSh.getRange(1, 1, rows.length, 2).setValues(rows);
    }

    return ContentService.createTextOutput(JSON.stringify({ok: true, saved: savedCount}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ok: false, error: err.message}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

// onEdit 트리거: aspro_match 시트에서 A열(사방넷코드) 입력 시
// C열(상품명)을 사방넷단품원본 시트에서 자동으로 조회하여 채움
// ──────────────────────────────────────────────
function onEdit(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = e.range.getSheet();
  if (sheet.getName() !== 'aspro_match') return;

  const editedRow = e.range.getRow();
  const editedCol = e.range.getColumn();
  if (editedRow === 1) return; // 헤더행 무시

  // A열 또는 B열 편집 시 상품명 자동완성
  if (editedCol !== 1 && editedCol !== 2) return;

  const sabCode = sheet.getRange(editedRow, 1).getValue();
  if (!sabCode || String(sabCode).trim() === '') return;

  // C열이 이미 채워져 있으면 덮어쓰지 않음
  const existingName = sheet.getRange(editedRow, 3).getValue();
  if (existingName && String(existingName).trim() !== '') return;

  // 사방넷단품원본 시트에서 상품명 조회
  const sabSh = ss.getSheetByName('사방넷단품원본');
  if (!sabSh) return;

  const sabData = sabSh.getDataRange().getValues();
  if (sabData.length < 2) return;

  const header = sabData[0].map(h => String(h).trim());
  const codeIdx = header.indexOf('사방넷상품코드') >= 0 ? header.indexOf('사방넷상품코드') : 0;
  const nameIdx = header.indexOf('상품명') >= 0 ? header.indexOf('상품명') : 2;

  const targetCode = String(sabCode).trim();
  for (let i = 1; i < sabData.length; i++) {
    const rowCode = String(sabData[i][codeIdx] || '').trim();
    if (rowCode === targetCode) {
      const productName = String(sabData[i][nameIdx] || '').trim();
      if (productName) {
        sheet.getRange(editedRow, 3).setValue(productName);
      }
      break;
    }
  }
}

// 일괄 상품명 채우기: aspro_match 시트의 빈 C열을 모두 채움
// 구글시트에서 직접 실행 가능 (메뉴 → 실행)
// ──────────────────────────────────────────────
function fillProductNames() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const matchSh = ss.getSheetByName('aspro_match');
  if (!matchSh) { SpreadsheetApp.getUi().alert('aspro_match 시트를 찾을 수 없습니다.'); return; }

  const sabSh = ss.getSheetByName('사방넷단품원본');
  if (!sabSh) { SpreadsheetApp.getUi().alert('사방넷단품원본 시트를 찾을 수 없습니다.'); return; }

  // 사방넷 코드 → 상품명 맵 빌드
  const sabData = sabSh.getDataRange().getValues();
  const header = sabData[0].map(h => String(h).trim());
  const codeIdx = header.indexOf('사방넷상품코드') >= 0 ? header.indexOf('사방넷상품코드') : 0;
  const nameIdx = header.indexOf('상품명') >= 0 ? header.indexOf('상품명') : 2;

  const sabNameMap = {};
  for (let i = 1; i < sabData.length; i++) {
    const c = String(sabData[i][codeIdx] || '').trim();
    const n = String(sabData[i][nameIdx] || '').trim();
    if (c && n) sabNameMap[c] = n;
  }

  // aspro_match 헤더 확인 및 보정
  const matchData = matchSh.getDataRange().getValues();
  const mHeader = matchData[0].map(h => String(h).trim());

  // 컬럼 수가 4개 이하면 헤더를 5컬럼으로 재설정
  if (mHeader.length < 3 || mHeader[2] !== '상품명') {
    matchSh.getRange(1, 1, 1, 5).setValues([['사방넷코드','김포물류품번','상품명','메모','등록일']]);
  }

  let filled = 0;
  for (let i = 1; i < matchData.length; i++) {
    const sabCode = String(matchData[i][0] || '').trim();
    if (!sabCode) continue;
    const existingName = String(matchData[i][2] || '').trim();
    if (existingName) continue; // 이미 있으면 스킵
    const name = sabNameMap[sabCode];
    if (name) {
      matchSh.getRange(i + 1, 3).setValue(name);
      filled++;
    }
  }

  SpreadsheetApp.getUi().alert(`✅ 상품명 채우기 완료: ${filled}건 업데이트`);
}
